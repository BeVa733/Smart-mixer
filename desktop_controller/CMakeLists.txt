cmake_minimum_required(VERSION 3.16)
project(smart_mixer_desktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../display_controller/.pio/libdeps/crowpanel_rev3/lvgl" CACHE PATH "LVGL 9.2.2 source")
if(NOT EXISTS "${LVGL_DIR}/lvgl.h")
  message(FATAL_ERROR "LVGL missing. Run pio pkg install -d display_controller or specify -DLVGL_DIR=...")
endif()
file(READ "${LVGL_DIR}/lv_version.h" LV_VERSION)
if(NOT LV_VERSION MATCHES "LVGL_VERSION_MINOR 2" OR NOT LV_VERSION MATCHES "LVGL_VERSION_PATCH 2" OR NOT LV_VERSION MATCHES "LVGL_VERSION_MAJOR 9")
  message(FATAL_ERROR "Use LVGL 9.2.2, matching CrowPanel")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" "${CMAKE_BINARY_DIR}/lvgl")
get_target_property(LVGL_SOURCES lvgl SOURCES)
list(FILTER LVGL_SOURCES EXCLUDE REGEX "/lv_sdl_keyboard\\.c$")
set_property(TARGET lvgl PROPERTY SOURCES ${LVGL_SOURCES})
target_sources(lvgl PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/SdlKeyboard.cpp")
target_link_libraries(lvgl PUBLIC PkgConfig::SDL2 m)
add_library(smart_mixer_ui ../shared/SmartMixerUi/src/MixerUi.cpp ../shared/SmartMixerUi/src/NumericInput.cpp ../shared/SmartMixerUi/src/mixer_font_16.c
            ../shared/SmartMixerUi/src/mixer_font_title.c ../shared/SmartMixerUi/src/menu_images.c)
target_include_directories(smart_mixer_ui PUBLIC ../shared/SmartMixerUi/src)
target_link_libraries(smart_mixer_ui PUBLIC lvgl)
add_executable(smart_mixer src/main.cpp src/UsbController.cpp src/RecipeStore.cpp)
target_compile_options(smart_mixer PRIVATE -Wall -Wextra -Werror)
target_link_libraries(smart_mixer PRIVATE smart_mixer_ui)
target_include_directories(smart_mixer_ui PUBLIC ../shared/SmartMixerControl/src)
enable_testing()
add_executable(calibration_tests src/test_calibration.cpp)
target_include_directories(calibration_tests PRIVATE src/calibration_stubs ../motion_controller/include)
target_compile_options(calibration_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME calibration_eeprom COMMAND calibration_tests)
add_executable(recipe_tests src/test_recipes.cpp src/RecipeStore.cpp)
target_link_libraries(recipe_tests PRIVATE smart_mixer_ui)
target_compile_options(recipe_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME recipes COMMAND recipe_tests)
add_executable(input_tests src/test_input.cpp)
target_link_libraries(input_tests PRIVATE smart_mixer_ui)
target_compile_options(input_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME input_keyboard COMMAND input_tests)
set_tests_properties(input_keyboard PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
add_executable(usb_tests src/test_usb.cpp src/UsbController.cpp)
target_link_libraries(usb_tests PRIVATE smart_mixer_ui util)
target_compile_options(usb_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME usb_pty COMMAND usb_tests)
add_test(NAME ui_smoke COMMAND smart_mixer --smoke-test)
set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
